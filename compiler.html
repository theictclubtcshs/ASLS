// ============================================ // GET FILE CONTENT (for section extraction) // ============================================ function getFileContent(fileId) { try { const file = DriveApp.getFileById(fileId); const content = file.getBlob().getDataAsString(); return { success: true, content: content, name: file.getName() }; } catch(err) { return { success: false, error: err.toString() }; } } // ============================================ // IMPROVED COMPILATION - EXTRACTS ALL DATA // ============================================ function compileAttendanceFromFiles(fileIds, filters) { try { const allRecords = []; let processedFiles = 0; // Process each file fileIds.forEach(fileId => { try { const file = DriveApp.getFileById(fileId); const content = file.getBlob().getDataAsString(); const fileName = file.getName(); // Extract records from HTML const records = extractAllRecordsFromHTML(content, fileName); allRecords.push(...records); processedFiles++; } catch(e) { console.error('Error processing file:', fileId, e); } }); // Apply filters let filtered = allRecords; if (filters.grade && filters.grade !== 'all') { filtered = filtered.filter(r => r.grade == filters.grade); } if (filters.section && filters.section !== 'all') { filtered = filtered.filter(r => r.section === filters.section); } // Time filter (extract from timestamp) if (filters.time && filters.time !== 'all') { filtered = filtered.filter(r => { if (!r.timestamp) return true; const time = r.timestamp.toLowerCase(); return time.includes(filters.time.toLowerCase()); }); } return { success: true, data: filtered, totalFiles: processedFiles, totalRecords: allRecords.length }; } catch(err) { return { success: false, error: err.toString() }; } } // ============================================ // EXTRACT ALL RECORDS FROM HTML // ============================================ function extractAllRecordsFromHTML(html, sourceFile) { const records = []; try { // Find all tables const tableRegex = //g; let tableMatch; while ((tableMatch = tableRegex.exec(html)) !== null) { const tableHTML = tableMatch[0]; // Check if this is a data table (look for headers) if (tableHTML.includes('Name') && tableHTML.includes('LRN')) { // Extract rows const rowRegex = /[\s\S]*?<\/tr>/g; let rowMatch; let rowIndex = 0; while ((rowMatch = rowRegex.exec(tableHTML)) !== null) { const rowHTML = rowMatch[0]; // Skip header rows if (rowHTML.includes('th') || rowIndex === 0) { rowIndex++; continue; } // Extract all TD cells const tdRegex = /]*>([\s\S]*?)<\/td>/g; const cells = []; let tdMatch; while ((tdMatch = tdRegex.exec(rowHTML)) !== null) { let cellContent = tdMatch[1].trim(); // Remove HTML tags cellContent = cellContent.replace(/<[^>]*>/g, '').trim(); cells.push(cellContent); } // If we have at least 6 cells, treat as a record if (cells.length >= 6) { const record = { timestamp: cells[0] || null, photo: cells[1] || null, name: cells[2] || null, lrn: cells[3] || null, sex: cells[4] || null, grade: cells[5] || null, section: cells[6] || null, action: cells[7] || null, sourceFile: sourceFile || null }; // Clean up section - handle cases like "7-A", "Grade 7 - A", etc. if (record.section) { // Try to extract just the section letter/number const sectionMatch = record.section.match(/[A-Za-z0-9]+$/); if (sectionMatch) { record.section = sectionMatch[0]; } } records.push(record); } rowIndex++; } } } // If no records found with table parsing, try alternative method if (records.length === 0) { // Look for pattern: Name, LRN, Grade, Section, Action in text const recordRegex = /([A-Za-z\s,.'-]+?)\s*[|,]\s*(\d{10,12})\s*[|,]\s*(\d+)\s*[|,]\s*([A-Za-z0-9-]+)\s*[|,]\s*(Log in|Log out)/gi; let match; while ((match = recordRegex.exec(html)) !== null) { records.push({ name: match[1].trim(), lrn: match[2].trim(), grade: match[3].trim(), section: match[4].trim(), action: match[5].trim(), sourceFile: sourceFile || null, timestamp: null, sex: null }); } } } catch(e) { console.error('Error extracting records:', e); } return records; } // ============================================ // MODIFIED doPost TO HANDLE ALL ACTIONS // ============================================ function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const action = payload.action || 'save'; switch(action) { case 'listFiles': return ContentService.createTextOutput(JSON.stringify(listAllAttendanceFiles())); case 'getFileContent': const fileResult = getFileContent(payload.fileId); return ContentService.createTextOutput(JSON.stringify(fileResult)); case 'compileAttendance': const compileResult = compileAttendanceFromFiles(payload.fileIds, payload.filters); return ContentService.createTextOutput(JSON.stringify(compileResult)); case 'save': default: // Your original save code here const data = payload.data; // ... existing save logic ... return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'File saved' })); } } catch(err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })); } } // ============================================ // doGet FOR GET REQUESTS // ============================================ function doGet(e) { const action = e.parameter.action; const fileId = e.parameter.fileId; if (action === 'listFiles') { return ContentService.createTextOutput(JSON.stringify(listAllAttendanceFiles())) .setMimeType(ContentService.MimeType.JSON); } if (action === 'getFileContent' && fileId) { const result = getFileContent(fileId); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'Invalid action' })); } // ============================================ // LIST ALL ATTENDANCE FILES // ============================================ function listAllAttendanceFiles() { try { const files = []; const folders = DriveApp.getFoldersByName('AttendanceASLS'); if (!folders.hasNext()) { return { success: false, error: 'AttendanceASLS folder not found' }; } const attendanceFolder = folders.next(); const yearFolders = attendanceFolder.getFolders(); while (yearFolders.hasNext()) { const yearFolder = yearFolders.next(); const year = yearFolder.getName(); const monthFolders = yearFolder.getFolders(); while (monthFolders.hasNext()) { const monthFolder = monthFolders.next(); const month = monthFolder.getName(); const dayFolders = monthFolder.getFolders(); while (dayFolders.hasNext()) { const dayFolder = dayFolders.next(); const day = dayFolder.getName(); const fileIterator = dayFolder.getFiles(); while (fileIterator.hasNext()) { const file = fileIterator.next(); const name = file.getName(); if (name.endsWith('.html')) { files.push({ id: file.getId(), name: name, path: `AttendanceASLS/${year}/${month}/${day}/${name}`, year: year, month: month, day: day, url: file.getUrl(), created: file.getDateCreated().toISOString() }); } } } } } return { success: true, files: files }; } catch(err) { return { success: false, error: err.toString() }; } }
